set(param_cli_sources param_command.cc param_cli_provider.cc)
add_library(ibmw_param_cli_provider SHARED ${param_cli_sources})
target_link_libraries(ibmw_param_cli_provider PRIVATE ibmw_cli_rpc_client ibmw_parameter_types)
set_target_properties(ibmw_param_cli_provider PROPERTIES OUTPUT_NAME "ibmw_param_cli"
  INSTALL_RPATH "$ORIGIN;$ORIGIN/../lib")
add_werror(ibmw_param_cli_provider)
set_property(GLOBAL APPEND PROPERTY IBMW_CLI_PROVIDER_TARGETS ibmw_param_cli_provider)
if(IBMW_INSTALL)
  install(TARGETS ibmw_param_cli_provider LIBRARY DESTINATION bin)
endif()

if(IBMW_BUILD_TESTS)
  add_executable(ibmw_param_cli_test param_command_test.cc ${param_cli_sources})
  target_link_libraries(ibmw_param_cli_test PRIVATE ibmw_cli_rpc_client ibmw_parameter_types GTest::gtest_main)
  add_werror(ibmw_param_cli_test)
  add_test(NAME ibmw_param_cli_test COMMAND ibmw_param_cli_test)
  set_tests_properties(ibmw_param_cli_test PROPERTIES TIMEOUT 30)

  if(IBMW_BUILD_PARAMETER_EXTENSION)
    add_executable(ibmw_parameter_server_fixture test/parameter_server_fixture.cc)
    target_link_libraries(ibmw_parameter_server_fixture PRIVATE ibmw::core::engine)
    add_werror(ibmw_parameter_server_fixture)
    add_dependencies(ibmw_param_cli_test ibmw_core_main ibmw_extensions_parameter_ext ibmw_parameter_server_fixture)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    set(param_test_backends)
    if(IBMW_BUILD_NET_EXTENSION)
      list(APPEND param_test_backends http)
      add_dependencies(ibmw_param_cli_test ibmw_extensions_net_ext)
    endif()
    if(IBMW_BUILD_DDS_EXTENSION)
      list(APPEND param_test_backends dds)
      add_dependencies(ibmw_param_cli_test ibmw_dds_extension)
    endif()
    if(IBMW_BUILD_GRPC_EXTENSION)
      list(APPEND param_test_backends grpc)
      add_dependencies(ibmw_param_cli_test ibmw_extensions_grpc_ext)
    endif()
    if(IBMW_BUILD_ZENOH_EXTENSION)
      list(APPEND param_test_backends zenoh)
      add_dependencies(ibmw_param_cli_test ibmw_extensions_zenoh_ext)
    endif()
    if(IBMW_BUILD_MQTT_EXTENSION)
      find_program(IBMW_PARAM_TEST_MOSQUITTO mosquitto)
      if(IBMW_PARAM_TEST_MOSQUITTO)
        list(APPEND param_test_backends mqtt)
        add_dependencies(ibmw_param_cli_test ibmw_extensions_mqtt_ext)
      endif()
    endif()
    foreach(backend IN LISTS param_test_backends)
      add_test(NAME ibmw_param_${backend}_test
        COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/scripts/tests/test_param_cli.py
          --cli "$<TARGET_FILE:ibmw_core_main>" --server "$<TARGET_FILE:ibmw_parameter_server_fixture>" --backend ${backend})
      set_tests_properties(ibmw_param_${backend}_test PROPERTIES TIMEOUT 180 RUN_SERIAL TRUE LABELS integration)
    endforeach()
  endif()
endif()
