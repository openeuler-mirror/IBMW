add_library(ibmw_cli_log_client STATIC log_client.cc)
target_link_libraries(ibmw_cli_log_client PUBLIC ibmw_cli_rpc_client ibmw_log_control_types)
target_include_directories(ibmw_cli_log_client PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_werror(ibmw_cli_log_client)

set(log_cli_sources log_command.cc log_cli_provider.cc)
add_library(ibmw_log_cli_provider SHARED ${log_cli_sources})
target_link_libraries(ibmw_log_cli_provider PRIVATE ibmw_cli_log_client)
set_target_properties(ibmw_log_cli_provider PROPERTIES OUTPUT_NAME "ibmw_log_cli"
  INSTALL_RPATH "$ORIGIN;$ORIGIN/../lib")
add_werror(ibmw_log_cli_provider)
set_property(GLOBAL APPEND PROPERTY IBMW_CLI_PROVIDER_TARGETS ibmw_log_cli_provider)
if(IBMW_INSTALL)
  install(TARGETS ibmw_log_cli_provider LIBRARY DESTINATION bin)
endif()

if(IBMW_BUILD_TESTS)
  add_executable(ibmw_log_cli_test log_command_test.cc ${log_cli_sources})
  target_link_libraries(ibmw_log_cli_test PRIVATE ibmw_cli_log_client GTest::gtest_main)
  add_werror(ibmw_log_cli_test)
  add_test(NAME ibmw_log_cli_test COMMAND ibmw_log_cli_test)
  set_tests_properties(ibmw_log_cli_test PROPERTIES TIMEOUT 30)

  if(IBMW_BUILD_LOG_CONTROL_EXTENSION)
    add_executable(ibmw_log_server_fixture test/log_server_fixture.cc)
    target_link_libraries(ibmw_log_server_fixture PRIVATE ibmw::core::engine)
    add_werror(ibmw_log_server_fixture)
    add_dependencies(ibmw_log_cli_test ibmw_core_main ibmw_extensions_log_control_ext ibmw_log_server_fixture)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    set(log_test_backends)
    if(IBMW_BUILD_NET_EXTENSION)
      list(APPEND log_test_backends http)
      add_dependencies(ibmw_log_cli_test ibmw_extensions_net_ext)
    endif()
    if(IBMW_BUILD_DDS_EXTENSION)
      list(APPEND log_test_backends dds)
      add_dependencies(ibmw_log_cli_test ibmw_dds_extension)
    endif()
    if(IBMW_BUILD_GRPC_EXTENSION)
      list(APPEND log_test_backends grpc)
      add_dependencies(ibmw_log_cli_test ibmw_extensions_grpc_ext)
    endif()
    if(IBMW_BUILD_ZENOH_EXTENSION)
      list(APPEND log_test_backends zenoh)
      add_dependencies(ibmw_log_cli_test ibmw_extensions_zenoh_ext)
    endif()
    if(IBMW_BUILD_MQTT_EXTENSION)
      find_program(IBMW_LOG_TEST_MOSQUITTO mosquitto)
      if(IBMW_LOG_TEST_MOSQUITTO)
        list(APPEND log_test_backends mqtt)
        add_dependencies(ibmw_log_cli_test ibmw_extensions_mqtt_ext)
      endif()
    endif()
    foreach(backend IN LISTS log_test_backends)
      add_test(NAME ibmw_log_${backend}_test
        COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/scripts/tests/test_log_cli.py
          --cli "$<TARGET_FILE:ibmw_core_main>" --server "$<TARGET_FILE:ibmw_log_server_fixture>" --backend ${backend})
      set_tests_properties(ibmw_log_${backend}_test PROPERTIES TIMEOUT 180 RUN_SERIAL TRUE LABELS integration)
    endforeach()
  endif()
endif()
