cmake_minimum_required(VERSION 3.20)
get_filename_component(IBMW_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../../../../../.." ABSOLUTE CACHE PATH "IBMW repository source root")
set(IBMW_OHOS_ROS2_PREFIX "" CACHE PATH "External full ROS2 installation prefix built for OHOS")
if(NOT EXISTS "${IBMW_SOURCE_DIR}/src/samples/cpp/transport_benchmark/topology_interop/ros2_endpoint.cc")
  message(FATAL_ERROR "IBMW_SOURCE_DIR is not an IBMW checkout")
endif()
if(NOT DEFINED CMAKE_TOOLCHAIN_FILE OR CMAKE_TOOLCHAIN_FILE STREQUAL "")
  set(CMAKE_TOOLCHAIN_FILE "${IBMW_SOURCE_DIR}/cmake/toolchains/ohos-arm64-v8a.cmake" CACHE FILEPATH "OHOS toolchain" FORCE)
endif()
if(IBMW_OHOS_ROS2_PREFIX STREQUAL "" OR
   NOT EXISTS "${IBMW_OHOS_ROS2_PREFIX}/share/rclcpp/cmake/rclcppConfig.cmake")
  message(FATAL_ERROR "IBMW_OHOS_ROS2_PREFIX must name a full OHOS ROS2 installation")
endif()
list(PREPEND CMAKE_PREFIX_PATH "${IBMW_OHOS_ROS2_PREFIX}")
list(PREPEND CMAKE_FIND_ROOT_PATH "${IBMW_OHOS_ROS2_PREFIX}")
project(ibmw_ohos_topology_ros2_standalone LANGUAGES CXX)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)

# Relocated OHOS exports can retain build-host paths. Remove only missing
# imported includes and Python generator links unused by this C++ endpoint.
get_property(_imported_targets DIRECTORY PROPERTY IMPORTED_TARGETS)
foreach(_target IN LISTS _imported_targets)
  get_target_property(_includes "${_target}" INTERFACE_INCLUDE_DIRECTORIES)
  if(_includes AND NOT _includes STREQUAL "_includes-NOTFOUND")
    set(_valid_includes)
    foreach(_include IN LISTS _includes)
      if(NOT IS_ABSOLUTE "${_include}" OR EXISTS "${_include}")
        list(APPEND _valid_includes "${_include}")
      endif()
    endforeach()
    set_target_properties("${_target}" PROPERTIES INTERFACE_INCLUDE_DIRECTORIES "${_valid_includes}")
  endif()
  get_target_property(_links "${_target}" INTERFACE_LINK_LIBRARIES)
  if(_links AND NOT _links STREQUAL "_links-NOTFOUND")
    set(_valid_links)
    foreach(_link IN LISTS _links)
      if(NOT _link MATCHES "__rosidl_generator_py$")
        list(APPEND _valid_links "${_link}")
      endif()
    endforeach()
    set_target_properties("${_target}" PROPERTIES INTERFACE_LINK_LIBRARIES "${_valid_links}")
  endif()
endforeach()
add_executable(ibmw_topology_ros2_endpoint "${IBMW_SOURCE_DIR}/src/samples/cpp/transport_benchmark/topology_interop/ros2_endpoint.cc" "${IBMW_SOURCE_DIR}/src/samples/cpp/transport_benchmark/topology_interop/topology_artifact.cc")
target_compile_features(ibmw_topology_ros2_endpoint PRIVATE cxx_std_20)
target_include_directories(ibmw_topology_ros2_endpoint PRIVATE "${IBMW_SOURCE_DIR}/src" "${IBMW_SOURCE_DIR}/src/samples/cpp/transport_benchmark/topology_interop")
target_link_libraries(ibmw_topology_ros2_endpoint PRIVATE rclcpp::rclcpp std_msgs::std_msgs__rosidl_generator_cpp std_msgs::std_msgs__rosidl_typesupport_cpp)
